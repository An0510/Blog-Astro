思路就是用BFS(广度优先遍历),用嵌套数组携带层级信息用于判断是哪一层

```js
var levelOrder = function (root) {
  if(!root) return []
  const res = []
  const queue = [[root, 0]]
  while(queue.length) {
    const [ node, level ] = queue.pop()
    if(res[level]) {
      res[level].push(node.val)
    } else {
      res[level] = [node.val]
    }
    if(node.right) queue.push([node.right, level + 1])
    if(node.left) queue.push([node.left, level + 1])
  }
  return res
};
```
